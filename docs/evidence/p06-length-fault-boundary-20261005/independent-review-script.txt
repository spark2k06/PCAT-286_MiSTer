import collections
import gzip
import hashlib
import json
from pathlib import Path, PurePosixPath
import subprocess
import tarfile

ROOT = Path(__file__).resolve().parents[2]
def sha(data): return hashlib.sha256(data).hexdigest()
def save(path, value): path.write_text(json.dumps(value, indent=2) + '\n')
def packet(key):
    folder = ROOT / 'docs/evidence' / key
    manifest = json.loads((folder / 'manifest.json').read_text())
    meta = manifest['archive'] if isinstance(manifest['archive'], dict) else {
        'path': manifest['archive'], 'bytes': manifest['bytes'], 'sha256': manifest['sha256']}
    raw = (folder / meta['path']).read_bytes()
    assert len(raw) == meta['bytes'] and sha(raw) == meta['sha256']
    with tarfile.open(folder / meta['path']) as archive:
        members = archive.getmembers()
        assert len({item.name for item in members}) == len(members)
        for item in members:
            name = PurePosixPath(item.name)
            assert not name.is_absolute() and '..' not in name.parts and item.isfile()
        content = {item.name: archive.extractfile(item).read() for item in members}
    return folder, manifest, meta, content

for source, description in receipt['source_manifest'].items():
    if source.startswith('baseline:'):
        original = source.split(':', 1)[1]
        data = subprocess.check_output(['git', 'show', receipt['baseline'] + ':' + original], cwd=ROOT)
        basename = 'spk286_control_decode_baseline.sv' if original.endswith('control_decode.sv') else 'spk286_decoder_baseline.sv'
        module = Path(original).stem
        renamed = data.replace(('module ' + module + ' (').encode(),
                               ('module ' + module + '_baseline (').encode())
        assert content['accepted/sources/' + basename] == renamed
    else:
        data = content['accepted/sources/' + source]
    assert len(data) == description['bytes'] and sha(data) == description['sha256']
for result in receipt['results']:
    assert result['pass'] in content[f"accepted/{result['label']}-{result['simulator']}-run.log"].decode()

    assert actual == receipt['generic_synthesis'][label]

    is_golden = 'kind' in run
    log_name = 'golden.log' if is_golden else ('candidate' if run['simulator'] == 'iverilog' else 'verilator') + '-' + str(run['profile']) + '.log'
    log = content[prefix + log_name].decode()
    assert 'P06_TARGET_HANDOFF_INTEGRATION_PASS' in log
    rows = {item[0]: [] for item in fields.values()}
    for line in log.splitlines():
        values = line.split(',')
        if values[0] not in fields: continue
        dest, names, hexadecimal = fields[values[0]]
        values = values[1:]
        parsed = [int(value, 16 if hexadecimal and index >= len(values)-hexadecimal else 10) for index,value in enumerate(values)]
        rows[dest].append(dict(zip(names, parsed, strict=True)))
